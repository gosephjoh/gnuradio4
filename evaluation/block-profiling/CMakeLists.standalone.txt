cmake_minimum_required(VERSION 3.20)
project(GR4Profile CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Python3 COMPONENTS Interpreter Development NumPy REQUIRED)
find_package(gnuradio4 REQUIRED)

target_link_libraries(gr4_profile PRIVATE gnuradio4::GrBasicBlocksShared)
set(CMAKE_CXX_STANDARD_LIBRARIES "${CMAKE_CXX_STANDARD_LIBRARIES} /usr/local/lib/libgnuradio-blocklib-core.so")
